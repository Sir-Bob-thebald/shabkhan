// lib/models/content.dart
enum ContentType { poem, story, essay }

class Content {
  final String title;
  final String body;
  final String? author;
  final String sourceUrl;
  final String sourceName;
  final ContentType type;
  final DateTime fetchedAt;

  Content({
    required this.title,
    required this.body,
    this.author,
    required this.sourceUrl,
    required this.sourceName,
    required this.type,
    required this.fetchedAt,
  });

  Map<String, dynamic> toJson() {
    return {
      'title': title,
      'body': body,
      'author': author,
      'sourceUrl': sourceUrl,
      'sourceName': sourceName,
      'type': type.name,
      'fetchedAt': fetchedAt.toIso8601String(),
    };
  }

  factory Content.fromJson(Map<String, dynamic> json) {
    return Content(
      title: json['title'],
      body: json['body'],
      author: json['author'],
      sourceUrl: json['sourceUrl'],
      sourceName: json['sourceName'],
      type: ContentType.values.firstWhere((e) => e.name == json['type']),
      fetchedAt: DateTime.parse(json['fetchedAt']),
    );
  }
}

// lib/models/source_config.dart
class SourceConfig {
  final String name;
  final String baseUrl;
  final ContentType contentType;
  final bool isEnabled;
  final Map<String, String> customHeaders;
  final String? selectorConfig; // CSS selectors or API endpoints

  SourceConfig({
    required this.name,
    required this.baseUrl,
    required this.contentType,
    this.isEnabled = true,
    this.customHeaders = const {},
    this.selectorConfig,
  });

  Map<String, dynamic> toJson() {
    return {
      'name': name,
      'baseUrl': baseUrl,
      'contentType': contentType.name,
      'isEnabled': isEnabled,
      'customHeaders': customHeaders,
      'selectorConfig': selectorConfig,
    };
  }

  factory SourceConfig.fromJson(Map<String, dynamic> json) {
    return SourceConfig(
      name: json['name'],
      baseUrl: json['baseUrl'],
      contentType: ContentType.values.firstWhere((e) => e.name == json['contentType']),
      isEnabled: json['isEnabled'] ?? true,
      customHeaders: Map<String, String>.from(json['customHeaders'] ?? {}),
      selectorConfig: json['selectorConfig'],
    );
  }
}

// lib/services/http_service.dart
import 'dart:convert';
import 'package:http/http.dart' as http;

class HttpService {
  static const Duration _timeout = Duration(seconds: 30);
  
  static Future<http.Response> get(String url, {Map<String, String>? headers}) async {
    try {
      final response = await http.get(
        Uri.parse(url),
        headers: {
          'User-Agent': 'ContentFetcher/1.0',
          ...?headers,
        },
      ).timeout(_timeout);
      
      return response;
    } catch (e) {
      throw Exception('Failed to fetch content: $e');
    }
  }
  
  static Future<Map<String, dynamic>> getJson(String url, {Map<String, String>? headers}) async {
    final response = await get(url, headers: headers);
    
    if (response.statusCode == 200) {
      return json.decode(response.body);
    } else {
      throw Exception('HTTP ${response.statusCode}: ${response.reasonPhrase}');
    }
  }
}

import 'package:html/parser.dart' as html_parser;
import '../../models/content.dart';
import '../../models/source_config.dart';
import '../content_source.dart';
import '../http_service.dart';
import 'dart:math';

class GanjoorSource extends ContentSource {
  GanjoorSource() : super(
    SourceConfig(
      name: 'Ganjoor',
      baseUrl: 'https://ganjoor.net',
      contentType: ContentType.poem,
    ),
  );

  @override
  Future<Content?> fetchRandomContent() async {
    try {
      // First, get a random poet page
      final poetsResponse = await HttpService.get('${config.baseUrl}/');
      final poetsDoc = html_parser.parse(poetsResponse.body);
      
      // Extract poet links - this is a simplified approach
      final poetLinks = poetsDoc.querySelectorAll('a[href*="/"]')
          .where((link) => link.attributes['href']?.contains('/') == true)
          .map((link) => link.attributes['href'])
          .where((href) => href != null && href.split('/').length >= 2)
          .toList();
      
      if (poetLinks.isEmpty) return null;
      
      // Select random poet
      final randomPoetUrl = poetLinks[Random().nextInt(poetLinks.length)];
      final fullPoetUrl = '${config.baseUrl}$randomPoetUrl';
      
      // Get poems from this poet
      final poetResponse = await HttpService.get(fullPoetUrl);
      final poetDoc = html_parser.parse(poetResponse.body);
      
      // Look for poem links
      final poemLinks = poetDoc.querySelectorAll('a[href*="/"]')
          .where((link) => link.attributes['href']?.contains('/') == true)
          .map((link) => link.attributes['href'])
          .where((href) => href != null)
          .toList();
      
      if (poemLinks.isEmpty) return null;
      
      // Get random poem
      final randomPoemUrl = poemLinks[Random().nextInt(poemLinks.length)];
      final fullPoemUrl = '${config.baseUrl}$randomPoemUrl';
      
      final poemResponse = await HttpService.get(fullPoemUrl);
      final poemDoc = html_parser.parse(poemResponse.body);
      
      // Extract poem content - adjust selectors based on actual site structure
      final titleElement = poemDoc.querySelector('h1, .poem-title, title');
      final contentElement = poemDoc.querySelector('.poem-text, .content, main');
      
      final title = titleElement?.text?.trim() ?? 'Untitled Poem';
      final body = contentElement?.text?.trim() ?? 'Content not found';
      
      return Content(
        title: title,
        body: body,
        sourceUrl: fullPoemUrl,
        sourceName: config.name,
        type: config.contentType,
        fetchedAt: DateTime.now(),
      );
      
    } catch (e) {
      print('Error fetching from Ganjoor: $e');
      return null;
    }
  }

  @override
  Future<List<Content>> fetchMultipleContent({int count = 5}) async {
    List<Content> contents = [];
    
    for (int i = 0; i < count; i++) {
      final content = await fetchRandomContent();
      if (content != null) {
        contents.add(content);
      }
    }
    
    return contents;
  }
}

import 'package:html/parser.dart' as html_parser;

import '../content_source.dart';
import '../http_service.dart';

class AppartusSource extends ContentSource {
  AppartusSource() : super(
    SourceConfig(
      name: 'Appartus',
      baseUrl: 'https://appartus.com',
      contentType: ContentType.story,
    ),
  );

  @override
  Future<Content?> fetchRandomContent() async {
    try {
      // Search for articles with "داستان" label
      final searchUrl = '${config.baseUrl}/?s=داستان';
      final response = await HttpService.get(searchUrl);
      final doc = html_parser.parse(response.body);
      
      // Find article links - this will need to be adjusted based on actual site structure
      final articleLinks = doc.querySelectorAll('a[href*="appartus.com"]')
          .where((link) => link.text.contains('داستان') || 
                         link.attributes['href']?.contains('story') == true)
          .map((link) => link.attributes['href'])
          .where((href) => href != null)
          .toList();
      
      if (articleLinks.isEmpty) {
        // Fallback: try to get any article
        final allLinks = doc.querySelectorAll('a[href*="appartus.com"]')
            .map((link) => link.attributes['href'])
            .where((href) => href != null)
            .toList();
        
        if (allLinks.isEmpty) return null;
        
        final randomUrl = allLinks[Random().nextInt(allLinks.length)];
        return await _fetchArticle(randomUrl!);
      }
      
      final randomUrl = articleLinks[Random().nextInt(articleLinks.length)];
      return await _fetchArticle(randomUrl!);
      
    } catch (e) {
      print('Error fetching from Appartus: $e');
      return null;
    }
  }
  
  Future<Content?> _fetchArticle(String url) async {
    try {
      final response = await HttpService.get(url);

      final title = doc.querySelector('h1, .title, .post-title')?.text?.trim() ?? 'Untitled Story';

        title: title,
        body: content,
        sourceUrl: url,
        sourceName: config.name,
        type: config.contentType,
        fetchedAt: DateTime.now(),

    } catch (e) {

// lib/services/content_manager.dart
import '../models/content.dart';
import '../models/source_config.dart';
import 'content_source.dart';
import 'sources/ganjoor_source.dart';
import 'sources/lesswrong_source.dart';
import 'sources/wikipedia_source.dart';
import 'sources/appartus_source.dart';
import 'dart:math';

class ContentManager {
  static final ContentManager _instance = ContentManager._internal();
  factory ContentManager() => _instance;
  ContentManager._internal();

  final List<ContentSource> _sources = [
    GanjoorSource(),
    LessWrongSource(), 
    WikipediaSource(),
    AppartusSource(),
  ];

  List<ContentSource> get sources => _sources;
  
  List<ContentSource> getSourcesByType(ContentType type) {
    return _sources.where((source) => source.config.contentType == type).toList();
  }
  
  Future<Content?> fetchRandomContent(ContentType type) async {
    final availableSources = getSourcesByType(type)
        .where((source) => source.config.isEnabled)
        .toList();
    
    if (availableSources.isEmpty) return null;
    
    final randomSource = availableSources[Random().nextInt(availableSources.length)];
    return await randomSource.fetchRandomContent();
  }
  
  Future<Map<ContentType, Content?>> fetchAllTypes() async {
    final results = <ContentType, Content?>{};
    
    for (final type in ContentType.values) {
      results[type] = await fetchRandomContent(type);
    }
    
    return results;
  }
  
  void addCustomSource(ContentSource source) {
    _sources.add(source);
  }
}

// lib/screens/home_screen.dart
import 'package:flutter/material.dart';
import '../models/content.dart';
import '../services/content_manager.dart';
import 'content_detail_screen.dart';
import 'settings_screen.dart';

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  final ContentManager _contentManager = ContentManager();
  Map<ContentType, Content?> _currentContent = {};
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    _fetchAllContent();
  }

  Future<void> _fetchAllContent() async {
    setState(() {
      _isLoading = true;
    });

    try {
      final content = await _contentManager.fetchAllTypes();
      setState(() {
        _currentContent = content;
      });
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Error fetching content: $e')),
      );
    } finally {
      setState(() {
        _isLoading = false;
      });
    }
  }

  Future<void> _fetchSpecificContent(ContentType type) async {
    setState(() {
      _isLoading = true;
    });

    try {
      final content = await _contentManager.fetchRandomContent(type);
      setState(() {
        _currentContent[type] = content;
      });
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Error fetching ${type.name}: $e')),
      );
    } finally {
      setState(() {
        _isLoading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Content Fetcher'),
        actions: [
          IconButton(
            icon: const Icon(Icons.settings),
            onPressed: () {
              Navigator.push(
                context,
                MaterialPageRoute(builder: (context) => const SettingsScreen()),
              );
            },
          ),
        ],
      ),
      body: _isLoading
          ? const Center(child: CircularProgressIndicator())
          : RefreshIndicator(
              onRefresh: _fetchAllContent,
              child: ListView(
                padding: const EdgeInsets.all(16.0),
                children: [
                  _buildContentCard(ContentType.poem),
                  const SizedBox(height: 16),
                  _buildContentCard(ContentType.story),
                  const SizedBox(height: 16),
                  _buildContentCard(ContentType.essay),
                  const SizedBox(height: 32),
                  ElevatedButton(
                    onPressed: _isLoading ? null : _fetchAllContent,
                    child: const Text('Fetch New Content'),
                  ),
                ],
              ),
            ),
    );
  }

  Widget _buildContentCard(ContentType type) {
    final content = _currentContent[type];
    final icon = _getIconForType(type);
    final color = _getColorForType(type);

    return Card(
      child: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              children: [
                Icon(icon, color: color),
                const SizedBox(width: 8),
                Text(
                  type.name.toUpperCase(),
                  style: Theme.of(context).textTheme.titleMedium?.copyWith(
                    color: color,
                    fontWeight: FontWeight.bold,
                  ),
                ),
                const Spacer(),
                IconButton(
                  icon: const Icon(Icons.refresh),
                  onPressed: () => _fetchSpecificContent(type),
                ),
              ],
            ),
            const SizedBox(height: 12),
            if (content != null) ...[
              Text(
                content.title,
                style: Theme.of(context).textTheme.titleLarge,
                maxLines: 2,
                overflow: TextOverflow.ellipsis,
              ),
              const SizedBox(height: 8),
              Text(
                content.body,
                style: Theme.of(context).textTheme.bodyMedium,
                maxLines: 4,
                overflow: TextOverflow.ellipsis,
              ),
              const SizedBox(height: 8),
              Row(
                children: [
                  Chip(
                    label: Text(content.sourceName),
                    backgroundColor: color.withOpacity(0.1),
                  ),
                  const Spacer(),
                  TextButton(
                    onPressed: () {
                      Navigator.push(
                        context,
                        MaterialPageRoute(
                          builder: (context) => ContentDetailScreen(content: content),
                        ),
                      );
                    },
                    child: const Text('Read More'),
                  ),
                ],
              ),
            ] else
              const Text('No content available. Try refreshing.'),
          ],
        ),
      ),
    );
  }

  IconData _getIconForType(ContentType type) {
    switch (type) {
      case ContentType.poem:
        return Icons.format_quote;
      case ContentType.story:
        return Icons.book;
      case ContentType.essay:
        return Icons.article;
    }
  }

  Color _getColorForType(ContentType type) {
    switch (type) {
      case ContentType.poem:
        return Colors.purple;
      case ContentType.story:
        return Colors.green;
      case ContentType.essay:
        return Colors.blue;
    }
  }
}
